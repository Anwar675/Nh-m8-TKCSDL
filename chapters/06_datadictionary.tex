\subsection{Quy tắc toàn vẹn của nhóm P1}

\begin{itemize}
    \item Mỗi email người dùng và mã đơn hàng là duy nhất; mã dịch vụ và mã gói chỉ cần duy nhất trong từng Film Lab.
    \item Dịch vụ, gói dịch vụ và đơn hàng liên quan phải thuộc cùng một Film Lab.
    \item \texttt{total\_amount} bằng tổng tạm tính cộng phí giao nhận và trừ giảm giá; kết quả không được âm.
    \item Giá trong \texttt{order\_items} là giá chốt phục vụ kiểm toán, không thay đổi khi giá danh mục được cập nhật.
    \item Không xóa cứng user, Film Lab hoặc dịch vụ nếu đã phát sinh đơn hàng; sử dụng trạng thái hoặc soft-delete.
\end{itemize}

\subsection{Quy tắc toàn vẹn của P4}

\begin{itemize}
    \item Mỗi file scan phải thuộc một cuộn phim và
    \texttt{scan\_files.film\_roll\_id} tham chiếu
    \texttt{film\_rolls.film\_roll\_id} của P3.

    \item Mỗi file scan có tối đa một bộ metadata.

    \item Một ảnh không được xuất hiện lặp lại trong cùng một album
    và một tag không được gắn lặp cho cùng một ảnh.

    \item Mỗi listing có tối đa một giao dịch; giá chốt trong giao dịch
    không thay đổi khi giá của listing được cập nhật.

    \item Mỗi giao dịch có tối đa một đánh giá.
    Người đánh giá phải là buyer của giao dịch và người được đánh giá
    phải là seller của listing.

    \item Rating chỉ nhận giá trị từ 1 đến 5.

    \item Không xóa cứng dữ liệu giao dịch hoặc đánh giá đã phát sinh;
    ưu tiên trạng thái nghiệp vụ để bảo toàn lịch sử.
\end{itemize}
