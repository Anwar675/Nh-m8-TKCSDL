
\chapter{THIẾT KẾ MỨC LOGIC VÀ CHUẨN HÓA}


\section{Miền dữ liệu dùng chung}

\begin{itemize}
    \item Định danh chính dùng \texttt{UUID}; thời gian dùng \texttt{TIMESTAMPTZ} theo UTC.
    \item Tiền tệ dùng \texttt{NUMERIC(12,2)} và phải lớn hơn hoặc bằng 0; mã tiền tệ dùng \texttt{CHAR(3)}.
    \item Email được chuẩn hóa chữ thường trước khi kiểm tra duy nhất; trạng thái dùng \texttt{VARCHAR} kèm \texttt{CHECK}, có thể chuyển thành PostgreSQL enum ở thiết kế vật lý.
    \item \texttt{created\_at} và \texttt{updated\_at} mặc định \texttt{CURRENT\_TIMESTAMP}; \texttt{deleted\_at} rỗng khi bản ghi còn hiệu lực.
\end{itemize}

\section{Quan hệ và quy tắc tham chiếu}

\begin{enumerate}
    \item Một user có nhiều role và một role có nhiều user thông qua \entity{user_roles}; tương tự, role và permission liên kết qua \entity{role_permissions}.
    \item Một owner có thể sở hữu nhiều Film Lab; khi user bị soft-delete, Film Lab không bị xóa theo mà phải chuyển quyền sở hữu hoặc khóa hoạt động.
    \item Một Film Lab có nhiều service và package. \entity{package_services} chỉ cho phép ghép service cùng tenant với package.
    \item Một customer tạo nhiều order; mỗi order thuộc đúng một Film Lab và có ít nhất một order item trước khi chuyển sang trạng thái \texttt{confirmed}.
    \item Không cascade delete từ user, Film Lab hoặc service sang order. Dữ liệu danh mục đã ngừng hoạt động được giữ để các đơn cũ vẫn truy vết được.
    \item \entity{order_items.unit_price} là snapshot giá, không tự cập nhật khi \entity{lab_services.base_price} thay đổi.
\end{enumerate}

\section{Kiểm tra chuẩn hóa đến 3NF}

Mỗi quan hệ có thuộc tính nguyên tố và không chứa nhóm lặp nên đạt 1NF. Các quan hệ có khóa ghép chỉ chứa thuộc tính phụ thuộc vào toàn bộ khóa: thời điểm gán phụ thuộc vào cặp user--role, số lượng dịch vụ phụ thuộc vào cặp package--service; vì vậy đạt 2NF. Thông tin user, Film Lab, service và package được tách riêng; order item chỉ giữ khóa tham chiếu cùng snapshot giá cần bảo toàn lịch sử, không lặp tên hoặc địa chỉ của thực thể khác. Không có thuộc tính không khóa phụ thuộc bắc cầu vào khóa chính, nên các quan hệ đạt 3NF.

Giá trị \entity{orders.total_amount} là dữ liệu dẫn xuất nhưng được giữ như snapshot tài chính để kiểm toán. Tính nhất quán của tổng tiền phải được kiểm tra trong transaction hoặc trigger ở thiết kế vật lý. \entity{service_options} dùng JSONB cho tùy chọn thay đổi theo loại dịch vụ; các thuộc tính cần tìm kiếm hoặc ràng buộc thường xuyên phải được tách thành quan hệ riêng khi chốt schema.

\section{Điểm tích hợp cần cross-review}

P3 cần xác nhận liên kết giữa \entity{order_items} và \entity{film_rolls}, state machine của order cùng lịch sử xử lý. P4 cần xác nhận đường truy vết từ film roll tới scan file và chủ sở hữu. Module Logistics/Payment cần dùng \entity{order_id} làm khóa tham chiếu, không lưu lặp thông tin customer hoặc Film Lab.

\subsection{Chuyển quan hệ và quy tắc tham chiếu P4}

\begin{enumerate}
    \item \entity{albums.user_id} tham chiếu \entity{users.user_id}.
    Hai quan hệ N--N được tách qua \entity{album_photos} và \entity{photo_tags};
    các cặp khóa ghép ngăn bản ghi liên kết trùng nhau.
    \item \entity{photo_metadata.scan_file_id} tham chiếu
    \entity{scan_files.scan_file_id} và có UNIQUE.
    Các khóa \entity{film_stock_id}, \entity{camera_model_id}, \entity{lens_id}
    tham chiếu các bảng danh mục tương ứng.
    \item \entity{products.product_category_id} tham chiếu
    \entity{product_categories.product_category_id};
    \entity{listings.product_id} tham chiếu \entity{products.product_id}.
    \entity{listings.seller_id} và \entity{marketplace_transactions.buyer_id}
    đều tham chiếu \entity{users.user_id}.
    \item \entity{seller_reviews.marketplace_transaction_id} tham chiếu
    \entity{marketplace_transactions.marketplace_transaction_id} và có UNIQUE.
    \entity{reviewer_id} và \entity{seller_id} đều tham chiếu
    \entity{users.user_id}, nhưng mang hai vai trò khác nhau.
    \item Không xóa dây chuyền dữ liệu giao dịch khi user hoặc listing ngừng
    hoạt động. Giá đã chốt ở \entity{agreed_price} được giữ để truy vết lịch sử.
\end{enumerate}

\subsection{Kiểm tra chuẩn hóa đến 3NF}

Các phụ thuộc hàm chính được xác định theo giả định nghiệp vụ của P4:
\begin{enumerate}
    \item \entity{scan_file_id} xác định các thuộc tính còn lại của
    \entity{scan_files}; \entity{photo_metadata_id} xác định các thuộc tính
    của \entity{photo_metadata}. Do \entity{scan_file_id} duy nhất trong
    \entity{photo_metadata}, khóa này cũng xác định bộ metadata tương ứng.
    \item \entity{album_id} xác định thông tin album;
    \entity{tag_id} xác định thông tin thẻ.
    Trong \entity{album_photos}, cặp
    (\entity{album_id}, \entity{scan_file_id}) xác định
    \entity{display_order} và \entity{added_at}.
    Trong \entity{photo_tags}, cặp
    (\entity{scan_file_id}, \entity{tag_id}) xác định
    \entity{created_at}.
    \item \entity{film_stock_id}, \entity{camera_model_id} và \entity{lens_id}
    lần lượt xác định thông tin của loại film, mẫu máy ảnh và ống kính.
    \item \entity{product_category_id} xác định thông tin danh mục;
    \entity{product_id} xác định thông tin sản phẩm;
    \entity{listing_id} xác định thông tin tin đăng.
    \item \entity{marketplace_transaction_id} xác định thông tin giao dịch.
    Với giả định mỗi listing có tối đa một giao dịch,
    \entity{listing_id} cũng là khóa ứng viên của
    \entity{marketplace_transactions}.
    \entity{seller_review_id} xác định thông tin đánh giá;
    \entity{marketplace_transaction_id} là khóa ứng viên của
    \entity{seller_reviews}.
\end{enumerate}

Các quan hệ có thuộc tính nguyên tố, không chứa nhóm lặp nên đáp ứng 1NF.
Phần lớn quan hệ dùng khóa đơn nên không có phụ thuộc bộ phận vào khóa ghép.
Hai quan hệ kết hợp chỉ chứa thuộc tính phụ thuộc vào toàn bộ cặp khóa,
vì vậy đáp ứng 2NF. Thông tin người dùng, danh mục, film, máy ảnh và ống kính
được tách riêng; các quan hệ giao dịch chỉ giữ khóa tham chiếu và dữ liệu
nghiệp vụ của chính giao dịch. Theo các phụ thuộc hàm đã nêu và giả định
mỗi thuộc tính mô tả phụ thuộc vào khóa của quan hệ tương ứng, mô hình
đáp ứng mục tiêu 3NF.

\entity{marketplace_transactions.agreed_price} là giá chốt tại thời điểm
giao dịch, được giữ để bảo toàn lịch sử và không tự cập nhật khi
\entity{listings.price} thay đổi. Đây là snapshot có chủ đích, không phải
sao chép tùy ý dữ liệu sản phẩm.

\subsection{Giả định nghiệp vụ và điểm tích hợp P4}

Để hoàn thành bản thiết kế tạm thời, P4 sử dụng các quy tắc sau:
\begin{enumerate}
    \item Mỗi \entity{scan_files} thuộc đúng một cuộn phim.
    \entity{film_roll_id} tham chiếu \entity{film_rolls.film_roll_id}.
    Chủ sở hữu ảnh được xác định qua đơn hàng của cuộn phim; không tạo
    thêm bảng users hoặc lưu lặp owner\_id trong scan\_files.
    Đường nối cụ thể tới order\_items cần đối chiếu với mô hình P3.
    \item Mỗi scan file có tối đa một metadata.
    UNIQUE \entity{photo_metadata.scan_file_id} bảo đảm ràng buộc này
    mà vẫn giữ PK \entity{photo_metadata_id} như ERD.
    \item Mỗi listing có tối đa một bản ghi giao dịch.
    Giao dịch có thể ở trạng thái chờ, hoàn thành hoặc hủy;
    không tạo nhiều bản ghi giao dịch cho cùng listing.
    UNIQUE \entity{marketplace_transactions.listing_id} bảo đảm
    cardinality 1--0..1. Nếu sau này cần lịch sử nhiều lần thử,
    nhóm phải điều chỉnh quy tắc này trước khi triển khai.
    \item Mỗi giao dịch có tối đa một đánh giá.
    \entity{seller_reviews.reviewer_id} phải là buyer của giao dịch,
    còn \entity{seller_reviews.seller_id} phải là seller của listing.
    Chỉ cho phép đánh giá giao dịch hoàn thành; người dùng không tự
    đánh giá chính mình. Rating nhận giá trị từ 1 đến 5.
    \item \entity{tags.tag_name} được chuẩn hóa chữ thường và duy nhất
    trong toàn hệ thống. \entity{products.condition} mô tả tình trạng
    của sản phẩm cụ thể, không phải tình trạng chung của một dòng máy.
    \item Không xóa dây chuyền dữ liệu giao dịch khi user hoặc listing
    ngừng hoạt động. Các kiểm tra xuyên nhiều bảng và quy tắc trạng thái
    sẽ được triển khai ở tầng nghiệp vụ hoặc cơ chế ràng buộc phù hợp
    trong giai đoạn thiết kế vật lý.
\end{enumerate}

Các quy tắc trên là phương án P4 chọn để thống nhất bản logic hiện tại,
không phải yêu cầu đã được nhóm trưởng xác nhận. Khi tích hợp với P3,
cần kiểm tra lại tên khóa cuộn phim và đường truy vết tới đơn hàng.

\subsection{Xac dinh phu thuoc ham va danh gia dang chuan}

\subsubsection{Tap phu thuoc ham (Functional Dependencies)}
\begin{itemize}
    \item ARTICLE\_CATEGORY: category\_id $\rightarrow$ category\_name, description, created\_at
    \item ARTICLE: article\_id $\rightarrow$ author\_id, category\_id, title, content, tags, view\_count, status, created\_at, updated\_at
    \item LAB\_REVIEW: review\_id $\rightarrow$ user\_id, lab\_id, rating, comment, created\_at
    \item COMMUNITY\_EVENT: event\_id $\rightarrow$ host\_id, title, description, location, event\_date, max\_participants, created\_at
    \item USER\_INTERACTION: interaction\_id $\rightarrow$ user\_id, target\_type, target\_id, interaction\_type, created\_at
    \item AI\_CHAT\_LOG: chat\_id $\rightarrow$ user\_id, user\_query, bot\_response, context\_retrieved, created\_at
\end{itemize}

\subsubsection{Xac dinh khoa va danh gia dang chuan 3NF / BCNF}
\begin{itemize}
    \item \textbf{Xac dinh khoa bang bao dong:} Tinh toan bao dong cho cac thuoc tinh khoa chinh cho thay moi thuoc tinh dinh danh deu xac dinh duoc toan bo cac thuoc tinh con lai trong bang. Do do, moi quan he deu co duy nhat mot khoa chinh don: category\_id, article\_id, review\_id, event\_id, interaction\_id, chat\_id.
    \item \textbf{Dang chuan 1 (1NF):} Tat ca cac cot du lieu deu mang gia tri nguyen to, khong co cot da tri hay lap lai.
    \item \textbf{Dang chuan 2 (2NF):} Do toan bo khoa chinh deu la khoa don nen khong ton tai bat ky phu thuoc ham bo phan nao vao mot phan cua khoa.
    \item \textbf{Dang chuan 3 (3NF) va BCNF:} Voi moi phu thuoc ham, ve trai deu la sieu khoa (khoa chinh). Khong co phu thuoc ham bac cau giua cac thuoc tinh khong khoa.
\end{itemize}

\textbf{Ket luan:} Toan bo cac bang trong phan he P5 dat dang chuan BCNF va 3NF, khong bi du thua du lieu va khong can thuc hien phan ra.
