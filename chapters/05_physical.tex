\chapter{THIẾT KẾ MỨC VẬT LÝ}

Chương này trình bày khái quát thiết kế vật lý của cơ sở dữ liệu trên
PostgreSQL, tập trung vào kiểu dữ liệu, khóa, ràng buộc và chỉ mục. Riêng P1
phụ trách nhóm bảng lõi gồm \texttt{users}, \texttt{roles},
\texttt{permissions}, \texttt{user\_roles}, \texttt{film\_labs},
\texttt{lab\_services}, \texttt{service\_packages}, \texttt{orders} và
\texttt{order\_items}, đồng thời dùng các quyết định này làm chuẩn để đối chiếu
với các module còn lại.

\subsection{Định hướng thiết kế vật lý}

\begin{itemize}
    \item Các thực thể nghiệp vụ chính dùng khóa chính \texttt{UUID}; bảng trung gian dùng khóa ghép để loại bỏ bản ghi trùng.
    \item Các cột thời gian dùng \texttt{TIMESTAMPTZ} theo UTC; bảng nghiệp vụ chính có \texttt{created\_at}, \texttt{updated\_at} và khi cần có thêm \texttt{deleted\_at}.
    \item Dữ liệu tiền tệ dùng \texttt{NUMERIC(12,2)}, dữ liệu văn bản ngắn dùng \texttt{VARCHAR}, mô tả dài dùng \texttt{TEXT}; riêng \texttt{JSONB} chỉ áp dụng cho tùy chọn linh hoạt của \texttt{order\_items}.
    \item Các khóa ngoại quan trọng ưu tiên \texttt{ON DELETE RESTRICT} nhằm bảo toàn lịch sử đơn hàng, dịch vụ và phân quyền.
\end{itemize}

\subsection{Ràng buộc và tối ưu chính}

\begin{itemize}
    \item Các thuộc tính định danh nghiệp vụ như email, mã vai trò, mã quyền, mã dịch vụ theo từng lab và mã đơn hàng được ràng buộc duy nhất.
    \item Các trạng thái nghiệp vụ được kiểm soát bằng \texttt{CHECK}; các cột giá tiền và số lượng phải thỏa điều kiện không âm hoặc lớn hơn 0.
    \item Chỉ mục tập trung vào các trường truy vấn thường xuyên như email đăng nhập, \texttt{lab\_id}, trạng thái Film Lab, trạng thái đơn hàng và lịch sử đơn theo khách hàng.
    \item Những quy tắc liên bảng như dịch vụ, gói dịch vụ và đơn hàng phải thuộc cùng một Film Lab được P1 kiểm tra khi review để đảm bảo tính nhất quán trên toàn schema.
\end{itemize}

\subsection{Một số câu lệnh minh họa}

Để minh họa cho cách triển khai vật lý của P1 trên PostgreSQL, có thể sử dụng
một số câu lệnh \texttt{CREATE TABLE} tiêu biểu như sau:

\begin{lstlisting}[caption={Minh họa DDL cho một số bảng lõi của P1}]
CREATE TABLE users (
    user_id UUID PRIMARY KEY,
    email VARCHAR(255) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(150) NOT NULL,
    status VARCHAR(20) NOT NULL CHECK (status IN ('active', 'inactive', 'suspended')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE film_labs (
    lab_id UUID PRIMARY KEY,
    owner_id UUID NOT NULL REFERENCES users(user_id),
    lab_name VARCHAR(150) NOT NULL,
    status VARCHAR(20) NOT NULL CHECK (status IN ('pending', 'approved', 'suspended', 'closed')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE orders (
    order_id UUID PRIMARY KEY,
    customer_id UUID NOT NULL REFERENCES users(user_id),
    lab_id UUID NOT NULL REFERENCES film_labs(lab_id),
    order_code VARCHAR(30) NOT NULL UNIQUE,
    total_amount NUMERIC(12,2) NOT NULL CHECK (total_amount >= 0),
    created_at TIMESTAMPTZ NOT NULL DEFAULT CURRENT_TIMESTAMP
);
\end{lstlisting}

\section{Thiết kế vật lý cho module P3 -- Order \& Processing Lifecycle}

Module P3 chịu trách nhiệm quản lý vòng đời đơn hàng, tiếp nhận cuộn
phim, theo dõi các công đoạn xử lý, lịch sử thay đổi trạng thái và các
yêu cầu nhận/giao. P3 sử dụng lại hai bảng lõi
\texttt{orders} và \texttt{order\_items} đã được định nghĩa ở P1, đồng
thời bổ sung các bảng phục vụ riêng cho quá trình xử lý phim.

Thiết kế vật lý của P3 tuân thủ các quy ước chung của hệ thống:
khóa chính sử dụng \texttt{UUID}, thời gian sử dụng
\texttt{TIMESTAMPTZ}, trạng thái sử dụng \texttt{VARCHAR} kết hợp
\texttt{CHECK}, và các khóa ngoại quan trọng ưu tiên
\texttt{ON DELETE RESTRICT} nhằm bảo toàn lịch sử xử lý.

\subsection{Thiết kế vật lý bảng \texttt{orders}}

Bảng \texttt{orders} là bảng trung tâm của vòng đời đơn hàng và đã
được định nghĩa trong nhóm bảng lõi của P1. P3 sử dụng lại bảng này,
không tạo thêm một bảng orders riêng.

\begin{itemize}

    \item \textbf{Khóa chính:} \texttt{order\_id} sử dụng kiểu
    \texttt{UUID}.

    \item \textbf{Khóa ngoại:}
    \texttt{customer\_id} tham chiếu đến
    \texttt{users(user\_id)} và \texttt{lab\_id} tham chiếu đến
    \texttt{film\_labs(lab\_id)}.

    \item \textbf{Mã đơn hàng:}
    \texttt{order\_code} sử dụng \texttt{VARCHAR(30)}, bắt buộc
    \texttt{NOT NULL} và có ràng buộc \texttt{UNIQUE}.

    \item \textbf{Trạng thái:}
    \texttt{status} sử dụng \texttt{VARCHAR(30)}. Giá trị phải tuân
    thủ state machine của hệ thống, gồm các trạng thái chính:
    \texttt{pending}, \texttt{confirmed}, \texttt{received},
    \texttt{processing}, \texttt{scanning}, \texttt{completed},
    \texttt{delivered} và trạng thái \texttt{cancelled} khi đơn bị
    hủy theo quy tắc nghiệp vụ.

    \item \textbf{Phương thức nhận/giao:}
    \texttt{fulfillment\_method} sử dụng \texttt{VARCHAR(20)} và chỉ
    nhận các giá trị \texttt{dropoff} hoặc \texttt{pickup}.

    \item \textbf{Dữ liệu tiền tệ:}
    \texttt{subtotal}, \texttt{delivery\_fee},
    \texttt{discount\_amount} và \texttt{total\_amount} sử dụng
    \texttt{NUMERIC(12,2)} và không được âm.

    \item \textbf{Thời gian:}
    \texttt{placed\_at}, \texttt{created\_at} và
    \texttt{updated\_at} sử dụng \texttt{TIMESTAMPTZ}.

    \item \textbf{Chỉ mục:} Tạo chỉ mục trên
    \texttt{customer\_id}, \texttt{lab\_id} và kết hợp
    \texttt{(status, created\_at)} để hỗ trợ truy vấn đơn hàng theo
    khách hàng, Film Lab và trạng thái hiện tại.

\end{itemize}


\subsection{Thiết kế vật lý bảng \texttt{order\_items}}

Bảng \texttt{order\_items} lưu các dịch vụ hoặc gói dịch vụ thuộc
một đơn hàng. Bảng này cũng đã được định nghĩa ở P1 và được P3 sử
dụng để liên kết đơn hàng với quá trình xử lý.

\begin{itemize}

    \item \textbf{Khóa chính:} \texttt{order\_item\_id} sử dụng
    kiểu \texttt{UUID}.

    \item \textbf{Khóa ngoại:}
    \texttt{order\_id} tham chiếu \texttt{orders(order\_id)};
    \texttt{service\_id} tham chiếu
    \texttt{lab\_services(service\_id)}; và
    \texttt{package\_id} tham chiếu
    \texttt{service\_packages(package\_id)} khi order item sử dụng
    package.

    \item \textbf{Số lượng:}
    \texttt{quantity} sử dụng \texttt{INTEGER} và phải lớn hơn 0.

    \item \textbf{Giá tại thời điểm đặt:}
    \texttt{unit\_price} sử dụng \texttt{NUMERIC(12,2)}, không âm và
    được giữ như snapshot giá để bảo toàn lịch sử giao dịch.

    \item \textbf{Tùy chọn dịch vụ:}
    \texttt{service\_options} sử dụng \texttt{JSONB} khi cần lưu các
    tùy chọn linh hoạt của dịch vụ.

    \item \textbf{Chỉ mục:} Tạo chỉ mục B-tree trên
    \texttt{order\_id} để tăng tốc truy vấn toàn bộ item của một đơn
    hàng. Các khóa ngoại \texttt{service\_id} và
    \texttt{package\_id} được lập chỉ mục khi có nhu cầu join thường
    xuyên.

\end{itemize}


\subsection{Thiết kế vật lý bảng \texttt{film\_rolls}}

Bảng \texttt{film\_rolls} quản lý từng cuộn phim thực tế được tiếp
nhận từ một đơn hàng. Đây là điểm liên kết quan trọng giữa vòng đời
đơn hàng của P3 và kho ảnh số của P4.

\begin{itemize}

    \item \textbf{Khóa chính:} \texttt{film\_roll\_id} sử dụng
    kiểu \texttt{UUID}.

    \item \textbf{Khóa ngoại:} \texttt{order\_id} sử dụng
    \texttt{UUID}, \texttt{NOT NULL}, tham chiếu đến
    \texttt{orders(order\_id)}.

    \item \textbf{Mã cuộn phim:}
    \texttt{roll\_code} sử dụng \texttt{VARCHAR(50)} để nhận diện
    từng cuộn phim. Có thể đặt \texttt{UNIQUE} nếu quy tắc nghiệp vụ
    yêu cầu mã cuộn duy nhất trong toàn hệ thống.

    \item \textbf{Loại phim:}
    \texttt{film\_type} sử dụng \texttt{VARCHAR(50)}.

    \item \textbf{Số frame:}
    \texttt{frame\_count} sử dụng \texttt{INTEGER}, cho phép
    \texttt{NULL} khi chưa xác định và nếu có giá trị thì phải lớn
    hơn hoặc bằng 0.

    \item \textbf{Thời gian xử lý:}
    \texttt{received\_at} và \texttt{scan\_completed\_at} sử dụng
    \texttt{TIMESTAMPTZ} và cho phép \texttt{NULL} trước khi các mốc
    xử lý tương ứng xảy ra.

    \item \textbf{Trạng thái:}
    \texttt{status} sử dụng \texttt{VARCHAR(30)} và được kiểm soát
    bằng \texttt{CHECK} theo miền trạng thái đã thống nhất.

    \item \textbf{Chỉ mục:} Tạo chỉ mục B-tree trên
    \texttt{order\_id} để truy vấn nhanh các cuộn phim thuộc một đơn
    hàng. Tạo thêm chỉ mục trên
    \texttt{(status, updated\_at)} để hỗ trợ màn hình tracking quá
    trình xử lý.

\end{itemize}


\subsection{Thiết kế vật lý bảng \texttt{processing\_stages}}

Bảng \texttt{processing\_stages} là bảng danh mục mô tả các công đoạn
xử lý phim.

\begin{itemize}

    \item \textbf{Khóa chính:}
    \texttt{processing\_stage\_id} sử dụng \texttt{UUID}.

    \item \textbf{Mã công đoạn:}
    \texttt{stage\_code} sử dụng \texttt{VARCHAR(50)}, bắt buộc
    \texttt{NOT NULL} và có ràng buộc \texttt{UNIQUE}.

    \item \textbf{Tên công đoạn:}
    \texttt{stage\_name} sử dụng \texttt{VARCHAR(100)} và không được
    rỗng.

    \item \textbf{Thứ tự công đoạn:}
    \texttt{stage\_order} sử dụng \texttt{INTEGER} và phải lớn hơn 0.

    \item \textbf{Trạng thái hoạt động:}
    \texttt{is\_active} sử dụng \texttt{BOOLEAN} để cho phép tạm
    ngừng một công đoạn mà không xóa dữ liệu lịch sử.

    \item \textbf{Chỉ mục:} Ngoài chỉ mục tự động của khóa chính và
    \texttt{stage\_code}, có thể tạo chỉ mục trên
    \texttt{stage\_order} để lấy danh sách công đoạn theo đúng thứ tự.

\end{itemize}


\subsection{Thiết kế vật lý bảng \texttt{stage\_logs}}

Bảng \texttt{stage\_logs} ghi nhận lịch sử xử lý của từng cuộn phim
tại từng công đoạn.

\begin{itemize}

    \item \textbf{Khóa chính:}
    \texttt{stage\_log\_id} sử dụng \texttt{UUID}.

    \item \textbf{Khóa ngoại:}
    \texttt{film\_roll\_id} tham chiếu
    \texttt{film\_rolls(film\_roll\_id)};
    \texttt{processing\_stage\_id} tham chiếu
    \texttt{processing\_stages(processing\_stage\_id)};
    \texttt{operator\_id} tham chiếu
    \texttt{users(user\_id)} và có thể nhận \texttt{NULL}.

    \item \textbf{Trạng thái:}
    \texttt{status} sử dụng \texttt{VARCHAR(30)} và được kiểm soát
    bằng \texttt{CHECK}.

    \item \textbf{Mốc thời gian:}
    \texttt{started\_at} và \texttt{completed\_at} sử dụng
    \texttt{TIMESTAMPTZ}. Nếu cả hai cùng có giá trị thì
    \texttt{completed\_at} phải lớn hơn hoặc bằng
    \texttt{started\_at}.

    \item \textbf{Toàn vẹn lịch sử:} Không sử dụng
    \texttt{ON DELETE CASCADE} từ \texttt{film\_rolls} hoặc
    \texttt{processing\_stages} sang lịch sử xử lý nếu điều đó làm
    mất khả năng truy vết.

    \item \textbf{Chỉ mục:} Tạo chỉ mục trên
    \texttt{film\_roll\_id} và kết hợp
    \texttt{(film\_roll\_id, created\_at)} để hiển thị timeline xử lý
    của một cuộn phim.

    \item Có thể tạo thêm chỉ mục trên
    \texttt{processing\_stage\_id} khi hệ thống thường xuyên truy vấn
    lịch sử theo từng công đoạn.

\end{itemize}


\subsection{Thiết kế vật lý bảng \texttt{order\_status\_history}}

Bảng \texttt{order\_status\_history} lưu toàn bộ lịch sử thay đổi
trạng thái của đơn hàng. Bảng này giúp hệ thống theo dõi chính xác
đơn hàng đã đi qua những trạng thái nào và thời điểm thay đổi.

\begin{itemize}

    \item \textbf{Khóa chính:}
    \texttt{order\_status\_history\_id} sử dụng \texttt{UUID}.

    \item \textbf{Khóa ngoại:}
    \texttt{order\_id} tham chiếu
    \texttt{orders(order\_id)} và
    \texttt{changed\_by} tham chiếu
    \texttt{users(user\_id)}.

    \item \textbf{Trạng thái cũ và mới:}
    \texttt{old\_status} sử dụng \texttt{VARCHAR(30)} và có thể
    \texttt{NULL} đối với bản ghi đầu tiên;
    \texttt{new\_status} sử dụng \texttt{VARCHAR(30)} và không được
    rỗng.

    \item \textbf{Thời điểm thay đổi:}
    \texttt{changed\_at} sử dụng \texttt{TIMESTAMPTZ} và không được
    rỗng.

    \item \textbf{Toàn vẹn lịch sử:} Ưu tiên
    \texttt{ON DELETE RESTRICT} đối với \texttt{order\_id} nhằm
    tránh xóa lịch sử khi order vẫn còn dữ liệu liên quan.

    \item \textbf{Chỉ mục quan trọng nhất:}
    tạo chỉ mục kết hợp
    \texttt{(order\_id, changed\_at DESC)}. Đây là chỉ mục phục vụ
    trực tiếp chức năng tracking lịch sử trạng thái của một order.

    \item Có thể tạo thêm chỉ mục trên
    \texttt{changed\_by} nếu thường xuyên cần truy vấn lịch sử theo
    người thực hiện.

\end{itemize}


\subsection{Thiết kế vật lý bảng \texttt{pickup\_delivery\_requests}}

Bảng \texttt{pickup\_delivery\_requests} quản lý các yêu cầu nhận
hoặc giao phim liên quan đến một đơn hàng.

\begin{itemize}

    \item \textbf{Khóa chính:}
    \texttt{request\_id} sử dụng \texttt{UUID}.

    \item \textbf{Khóa ngoại:}
    \texttt{order\_id} sử dụng \texttt{UUID}, bắt buộc
    \texttt{NOT NULL}, tham chiếu đến
    \texttt{orders(order\_id)}.

    \item \textbf{Loại yêu cầu:}
    \texttt{request\_type} sử dụng \texttt{VARCHAR(20)} và chỉ
    nhận các giá trị \texttt{pickup} hoặc \texttt{delivery}.

    \item \textbf{Trạng thái:}
    \texttt{status} sử dụng \texttt{VARCHAR(30)} và được kiểm soát
    bằng \texttt{CHECK}.

    \item \textbf{Địa chỉ:}
    \texttt{pickup\_address} và \texttt{delivery\_address} sử dụng
    \texttt{TEXT} và có thể rỗng tùy theo loại yêu cầu.

    \item \textbf{Thời gian:}
    \texttt{requested\_at}, \texttt{completed\_at},
    \texttt{created\_at} và \texttt{updated\_at} sử dụng
    \texttt{TIMESTAMPTZ}.

    \item \textbf{Ràng buộc nghiệp vụ:} Nếu
    \texttt{request\_type = 'pickup'} thì
    \texttt{pickup\_address} phải tồn tại; nếu
    \texttt{request\_type = 'delivery'} thì
    \texttt{delivery\_address} phải tồn tại. Khi
    \texttt{completed\_at} có giá trị thì phải lớn hơn hoặc bằng
    \texttt{requested\_at}.

    \item \textbf{Chỉ mục:} Tạo chỉ mục kết hợp
    \texttt{(order\_id, status)} để hỗ trợ truy vấn trạng thái
    yêu cầu nhận/giao của một đơn hàng.

    \item Có thể tạo thêm chỉ mục
    \texttt{(status, requested\_at)} để hỗ trợ màn hình quản lý các
    yêu cầu đang chờ xử lý theo thời gian.

\end{itemize}


\subsection{Chiến lược chỉ mục và tối ưu truy vấn cho P3}

P3 có workload chủ yếu phục vụ chức năng tracking theo thời gian
thực. Vì vậy chỉ mục được ưu tiên cho các trường thường xuyên xuất
hiện trong điều kiện \texttt{WHERE}, \texttt{JOIN} và sắp xếp theo
thời gian.

Các chỉ mục quan trọng gồm:

\begin{itemize}

    \item \texttt{orders(customer\_id)}: truy vấn danh sách đơn hàng
    của một khách hàng.

    \item \texttt{orders(lab\_id)}: truy vấn các đơn hàng thuộc một
    Film Lab.

    \item \texttt{orders(status, created\_at)}: lọc đơn hàng theo
    trạng thái và thời gian tạo.

    \item \texttt{film\_rolls(order\_id)}: tìm toàn bộ cuộn phim
    thuộc một đơn hàng.

    \item \texttt{film\_rolls(status, updated\_at)}: hỗ trợ màn hình
    tracking các cuộn phim đang được xử lý.

    \item \texttt{stage\_logs(film\_roll\_id, created\_at)}:
    lấy timeline xử lý của một cuộn phim.

    \item \texttt{order\_status\_history(order\_id, changed\_at DESC)}:
    lấy nhanh lịch sử trạng thái theo thứ tự mới nhất.

    \item \texttt{pickup\_delivery\_requests(order\_id, status)}:
    theo dõi yêu cầu nhận/giao của một đơn hàng.

\end{itemize}

Không tạo quá nhiều chỉ mục trên các bảng có tần suất ghi cao như
\texttt{stage\_logs} và \texttt{order\_status\_history}, vì mỗi lần
INSERT phải cập nhật thêm các cấu trúc chỉ mục. Do đó chỉ giữ các chỉ
mục phục vụ trực tiếp các truy vấn tracking quan trọng.


\subsection{Toàn vẹn tham chiếu và chính sách xóa P3}

Các bảng lịch sử của P3 phải ưu tiên bảo toàn dữ liệu. Vì vậy các khóa
ngoại quan trọng tham chiếu đến \texttt{orders},
\texttt{film\_rolls}, \texttt{processing\_stages} và
\texttt{users} không nên sử dụng cascade delete nếu việc đó có thể làm
mất dữ liệu lịch sử.

Đối với \texttt{orders}, khi đơn hàng đã phát sinh dữ liệu xử lý,
không nên cho phép xóa vật lý trực tiếp. Thay vào đó, hệ thống có thể
sử dụng trạng thái hoặc cơ chế soft delete phù hợp với thiết kế chung.

Đối với \texttt{order\_status\_history} và \texttt{stage\_logs}, dữ
liệu phải được giữ lại để phục vụ kiểm toán và truy vết quá trình xử lý.

\section{Thiết kế vật lý cho module P4}

Module P4 gồm hai nhóm dữ liệu chính là Kho ảnh phim số và Chợ giao dịch.
Thiết kế vật lý sử dụng lại các quy ước chung của hệ thống: khóa chính dùng
\texttt{UUID}, thời gian dùng \texttt{TIMESTAMPTZ}, giá tiền dùng
\texttt{NUMERIC(12,2)} và các khóa ngoại phải được lập chỉ mục khi thường
xuyên tham gia truy vấn.

\subsection{Kho ảnh phim số}

Bảng \texttt{scan\_files} sử dụng \texttt{scan\_file\_id UUID} làm khóa chính.
\texttt{film\_roll\_id UUID} là khóa ngoại tham chiếu đến cuộn phim của module
P3 và không được rỗng. Đường dẫn ảnh được lưu trong
\texttt{storage\_url TEXT}, không lưu trực tiếp dữ liệu nhị phân của ảnh
trong cơ sở dữ liệu. Tên file dùng \texttt{VARCHAR(255)}, định dạng file dùng
\texttt{VARCHAR(20)}, kích thước file dùng \texttt{BIGINT} và phải lớn hơn
hoặc bằng 0.

Bảng \texttt{photo\_metadata} dùng
\texttt{photo\_metadata\_id UUID} làm khóa chính.
\texttt{scan\_file\_id UUID} là khóa ngoại đồng thời có ràng buộc
\texttt{UNIQUE} để bảo đảm mỗi file scan có tối đa một bộ metadata.
Các khóa \texttt{film\_stock\_id}, \texttt{camera\_model\_id} và
\texttt{lens\_id} là khóa ngoại và có thể rỗng nếu thông tin ảnh chưa đầy đủ.

Bảng \texttt{albums} dùng \texttt{album\_id UUID} làm khóa chính và
\texttt{user\_id UUID} tham chiếu người sở hữu album.
Quan hệ nhiều--nhiều giữa album và file scan được triển khai bằng
\texttt{album\_photos}, với khóa chính ghép
(\texttt{album\_id}, \texttt{scan\_file\_id}).

Bảng \texttt{tags} sử dụng \texttt{tag\_id UUID} làm khóa chính.
\texttt{tag\_name VARCHAR(100)} được chuẩn hóa chữ thường và có ràng buộc
duy nhất. Quan hệ nhiều--nhiều giữa file scan và tag được triển khai bằng
\texttt{photo\_tags}, sử dụng khóa chính ghép
(\texttt{scan\_file\_id}, \texttt{tag\_id}).

Các bảng danh mục \texttt{film\_stocks}, \texttt{camera\_models} và
\texttt{lenses} đều sử dụng khóa chính UUID. Các thuộc tính mô tả như brand,
model, film format, focal length và mount type sử dụng \texttt{VARCHAR}
với độ dài phù hợp nhằm hạn chế dữ liệu không cần thiết.

\subsection{Chợ giao dịch}

Bảng \texttt{product\_categories} dùng
\texttt{product\_category\_id UUID} làm khóa chính.
Tên danh mục dùng \texttt{VARCHAR(150)} và không được rỗng.

Bảng \texttt{products} dùng \texttt{product\_id UUID} làm khóa chính và
\texttt{product\_category\_id UUID} làm khóa ngoại.
Tên sản phẩm dùng \texttt{VARCHAR(150)}, thương hiệu và model dùng
\texttt{VARCHAR}, còn mô tả dài sử dụng \texttt{TEXT}.

Bảng \texttt{listings} sử dụng \texttt{listing\_id UUID} làm khóa chính.
\texttt{product\_id} tham chiếu sản phẩm và \texttt{seller\_id} tham chiếu
\texttt{users.user\_id}. Giá bán dùng \texttt{NUMERIC(12,2)} và phải lớn hơn
hoặc bằng 0. Trạng thái tin đăng sử dụng \texttt{VARCHAR(20)} kết hợp
ràng buộc \texttt{CHECK}.

Bảng \texttt{marketplace\_transactions} sử dụng
\texttt{marketplace\_transaction\_id UUID} làm khóa chính.
\texttt{listing\_id UUID} là khóa ngoại và có ràng buộc \texttt{UNIQUE},
phù hợp với quy tắc mỗi tin đăng có tối đa một giao dịch.
\texttt{buyer\_id UUID} tham chiếu \texttt{users.user\_id}.
Giá chốt dùng \texttt{NUMERIC(12,2)} và được giữ như snapshot lịch sử.

Bảng \texttt{seller\_reviews} sử dụng
\texttt{seller\_review\_id UUID} làm khóa chính.
\texttt{marketplace\_transaction\_id UUID} là khóa ngoại có
\texttt{UNIQUE}, bảo đảm mỗi giao dịch có tối đa một đánh giá.
\texttt{reviewer\_id} và \texttt{seller\_id} đều tham chiếu
\texttt{users.user\_id}. Điểm đánh giá dùng \texttt{INTEGER} và được giới hạn
từ 1 đến 5 bằng ràng buộc \texttt{CHECK}.

\subsection{Chỉ mục và tối ưu truy vấn cho P4}

Các khóa ngoại thường xuyên được sử dụng để join như
\texttt{film\_roll\_id}, \texttt{user\_id}, \texttt{product\_category\_id},
\texttt{product\_id}, \texttt{seller\_id} và \texttt{buyer\_id} cần được
lập chỉ mục.

Đối với Chợ giao dịch, nên có chỉ mục kết hợp trên
(\texttt{status}, \texttt{created\_at}) của \texttt{listings} để hỗ trợ
lọc các tin đang hoạt động và sắp xếp theo thời gian mới nhất.
Có thể lập thêm chỉ mục cho giá nếu chức năng tìm kiếm thường xuyên lọc
theo khoảng giá.

Đối với Kho ảnh số, chỉ mục trên \texttt{scan\_files.film\_roll\_id},
\texttt{albums.user\_id} và các khóa ngoại của
\texttt{photo\_metadata} giúp tăng tốc truy vấn ảnh theo cuộn phim,
người sở hữu và metadata.

\subsection{Lưu trữ và toàn vẹn dữ liệu P4}

File ảnh scan được lưu ở object storage, cơ sở dữ liệu chỉ lưu URI và
metadata cần thiết. Việc xóa bản ghi không được tự động xóa file ảnh vật lý
nếu chưa hoàn tất quy trình kiểm tra quyền sở hữu và tham chiếu.

Dữ liệu giao dịch và đánh giá cần được giữ lại để truy vết lịch sử.
Không cascade delete từ \texttt{users}, \texttt{listings} hoặc sản phẩm
sang các giao dịch đã phát sinh. Các trạng thái nghiệp vụ phải được kiểm tra
ở tầng ứng dụng kết hợp với ràng buộc cơ sở dữ liệu.
\section{Thiết kế vật lý cho module P5 -- Knowledge \& AI Integration}

Module P5 chịu trách nhiệm xây dựng kho tri thức nhiếp ảnh film, kết nối cộng đồng thông qua sự kiện, thu thập đánh giá dịch vụ Film Lab, theo dõi hành vi tương tác và lưu vết hội thoại với trợ lý AI. Thiết kế mức vật lý cho module P5 được chuyển đổi từ lược đồ quan hệ đã chuẩn hóa ở Chương 4, đồng thời tuân thủ toàn bộ các nguyên tắc thiết kế chung của hệ thống trên hệ quản trị cơ sở dữ liệu PostgreSQL.

Các quyết định thiết kế cốt lõi bao gồm:
\begin{itemize}
    \item \textbf{Khóa chính:} Toàn bộ các thực thể nghiệp vụ độc lập sử dụng định danh duy nhất \texttt{UUID} (v4) nhằm đảm bảo tính phân tán, tránh đoán trước khóa và an toàn khi tích hợp với các hệ thống phân tán.
    \item \textbf{Kiểu thời gian:} Mọi thuộc tính mốc thời gian đều sử dụng \texttt{TIMESTAMPTZ} (timestamp with time zone) để đồng bộ múi giờ quốc tế và đảm bảo độ chính xác cho việc sắp xếp dòng thời gian sự kiện.
    \item \textbf{Dữ liệu phi cấu trúc và văn bản dài:} Các trường nội dung bài viết, mã ngữ cảnh RAG và phản hồi AI sử dụng kiểu \texttt{TEXT} có cơ chế lưu trữ ngoài dòng (TOAST) của PostgreSQL nhằm tránh làm phình trang đĩa chính (heap page).
    \item \textbf{Chiến lược toàn vẹn tham chiếu:} Áp dụng nghiêm ngặt các tùy chọn \texttt{ON DELETE CASCADE}, \texttt{ON DELETE RESTRICT} và \texttt{ON DELETE SET NULL} tương ứng với từng mối quan hệ logic nghiệp vụ.
    \item \textbf{Tối ưu hóa truy vấn (Indexing):} Đánh chỉ mục B-tree trên tất cả các khóa ngoại, chỉ mục phức hợp (composite index) cho dữ liệu gợi ý và chỉ mục GIN hỗ trợ tìm kiếm toàn văn (Full-text Search).
\end{itemize}

\subsection{Thiết kế vật lý nhóm Cơ sở tri thức và Đánh giá}

\subsubsection{Thiết kế vật lý bảng \texttt{article\_categories}}

Bảng \texttt{article\_categories} đóng vai trò là bảng danh mục phân loại bài viết kiến thức, có tần suất ghi thấp nhưng tần suất đọc rất cao.

\begin{itemize}
    \item \textbf{Khóa chính:} Thuộc tính \texttt{category\_id} dùng kiểu \texttt{UUID} với giá trị mặc định sinh tự động \texttt{gen\_random\_uuid()}.
    \item \textbf{Thuộc tính nghiệp vụ:}
    \begin{itemize}
        \item \texttt{category\_name}: Sử dụng \texttt{VARCHAR(100)} đi kèm ràng buộc \texttt{NOT NULL} và \texttt{UNIQUE} để ngăn chặn trùng lặp danh mục.
        \item \texttt{description}: Dùng kiểu \texttt{TEXT}, cho phép \texttt{NULL} để người quản trị bổ sung phần diễn giải phạm vi chủ đề khi cần thiết.
        \item \texttt{created\_at}: Sử dụng \texttt{TIMESTAMPTZ} với giá trị mặc định là \texttt{CURRENT\_TIMESTAMP}.
    \end{itemize}
    \item \textbf{Chỉ mục:} Hệ thống tự động tạo chỉ mục B-tree duy nhất cho khóa chính và ràng buộc \texttt{UNIQUE} trên cột \texttt{category\_name}. Do số lượng danh mục bài viết không lớn, bảng này không yêu cầu phân vùng hay chỉ mục mở rộng.
\end{itemize}

\subsubsection{Thiết kế vật lý bảng \texttt{articles}}

Bảng \texttt{articles} là bảng nghiệp vụ trung tâm của kho tri thức, lưu trữ các bài viết chia sẻ kinh nghiệm, đánh giá máy ảnh/film và kỹ thuật phòng tối.

\begin{itemize}
    \item \textbf{Khóa chính:} \texttt{article\_id} dùng kiểu \texttt{UUID}.
    \item \textbf{Khóa ngoại và toàn vẹn dữ liệu:}
    \begin{itemize}
        \item \texttt{author\_id} (\texttt{UUID}, \texttt{NOT NULL}): Tham chiếu đến \texttt{users(user\_id)} thuộc module P1. Sử dụng chính sách \texttt{ON DELETE CASCADE} nhằm đảm bảo khi tài khoản tác giả bị xóa khỏi hệ thống, các bài viết liên quan sẽ được xử lý tương ứng theo quy định bảo mật.
        \item \texttt{category\_id} (\texttt{UUID}, \texttt{NOT NULL}): Tham chiếu đến \texttt{article\_categories(category\_id)}. Áp dụng chính sách \texttt{ON DELETE RESTRICT} để ngăn chặn việc xóa danh mục khi vẫn còn các bài viết trực thuộc.
    \end{itemize}
    \item \textbf{Thuộc tính nghiệp vụ và lưu trữ:}
    \begin{itemize}
        \item \texttt{title}: Sử dụng \texttt{VARCHAR(255)} có ràng buộc \texttt{NOT NULL}.
        \item \texttt{content}: Kiểu \texttt{TEXT} có kích thước biến thiên lớn. Thuộc tính này tự động kích hoạt cơ chế TOAST khi độ dài vượt quá ngưỡng kích thước trang chuẩn (thường là 2KB), giúp các truy vấn quét danh sách tiêu đề bài viết không bị suy giảm hiệu năng do phải đọc dữ liệu nội dung thô.
        \item \texttt{tags}: Sử dụng \texttt{VARCHAR(255)} lưu chuỗi từ khóa phân tách.
        \item \texttt{view\_count}: Dùng kiểu \texttt{INT}, mặc định \texttt{0}, đi kèm ràng buộc kiểm tra \texttt{CHECK (view\_count >= 0)}.
        \item \texttt{status}: Dùng \texttt{VARCHAR(20)} mặc định \texttt{'published'}, áp dụng \texttt{CHECK (status IN ('draft', 'published', 'archived'))} nhằm kiểm soát chặt chẽ vòng đời của bài viết.
        \item \texttt{created\_at} và \texttt{updated\_at}: Kiểu \texttt{TIMESTAMPTZ}, mặc định \texttt{CURRENT\_TIMESTAMP}. Cột \texttt{updated\_at} được cập nhật tự động thông qua cơ chế trigger khi có thay đổi nội dung.
    \end{itemize}
    \item \textbf{Chiến lược chỉ mục:}
    \begin{itemize}
        \item Chỉ mục B-tree trên \texttt{author\_id} nhằm tăng tốc độ tải trang danh sách bài viết cá nhân của tác giả.
        \item Chỉ mục B-tree trên \texttt{category\_id} phục vụ lọc bài viết theo từng phân mục cụ thể.
        \item Chỉ mục phức hợp \texttt{(status, created\_at DESC)} phục vụ cho câu truy vấn phổ biến nhất trên trang chủ cộng đồng: lấy danh sách các bài viết đã xuất bản mới nhất.
    \end{itemize}
\end{itemize}

\subsubsection{Thiết kế vật lý bảng \texttt{lab\_reviews}}

Bảng \texttt{lab\_reviews} lưu trữ các phản hồi, đánh giá thực tế của cộng đồng về chất lượng tráng scan của từng Film Lab.

\begin{itemize}
    \item \textbf{Khóa chính:} \texttt{review\_id} kiểu \texttt{UUID}.
    \item \textbf{Khóa ngoại:}
    \begin{itemize}
        \item \texttt{user\_id} (\texttt{UUID}, \texttt{NOT NULL}): Tham chiếu đến \texttt{users(user\_id)} với chính sách \texttt{ON DELETE CASCADE}.
        \item \texttt{lab\_id} (\texttt{UUID}, \texttt{NOT NULL}): Tham chiếu đến bảng phòng lab \texttt{film\_labs(lab\_id)} của module P1 với chính sách \texttt{ON DELETE CASCADE}.
    \end{itemize}
    \item \textbf{Thuộc tính kiểm tra và đánh giá:}
    \begin{itemize}
        \item \texttt{rating}: Kiểu \texttt{INT} có ràng buộc \texttt{CHECK (rating >= 1 AND rating <= 5)} đảm bảo thang điểm chuẩn 5 sao.
        \item \texttt{comment}: Kiểu \texttt{TEXT}, cho phép \texttt{NULL} trong trường hợp người dùng chỉ để lại số sao mà không viết nhận xét dài.
        \item \texttt{created\_at}: Kiểu \texttt{TIMESTAMPTZ}, mặc định \texttt{CURRENT\_TIMESTAMP}.
    \end{itemize}
    \item \textbf{Chiến lược chỉ mục:} Thiết lập chỉ mục B-tree trên \texttt{lab\_id} để tối ưu hóa truy vấn tính điểm trung bình (AVG rating) và hiển thị danh sách đánh giá của từng Lab.
\end{itemize}

\subsection{Thiết kế vật lý nhóm Sự kiện, Tương tác và Trí tuệ nhân tạo}

\subsubsection{Thiết kế vật lý bảng \texttt{community\_events}}

Bảng \texttt{community\_events} quản lý dữ liệu lịch trình của các buổi offline, photowalk săn ảnh film và workshop kỹ thuật cộng đồng.

\begin{itemize}
    \item \textbf{Khóa chính:} \texttt{event\_id} kiểu \texttt{UUID}.
    \item \textbf{Khóa ngoại:} \texttt{host\_id} (\texttt{UUID}, \texttt{NOT NULL}) tham chiếu đến tài khoản người dùng đại diện tổ chức \texttt{users(user\_id)} với chính sách \texttt{ON DELETE CASCADE}.
    \item \textbf{Thuộc tính sự kiện:}
    \begin{itemize}
        \item \texttt{title} (\texttt{VARCHAR(255)}, \texttt{NOT NULL}): Tên buổi sự kiện.
        \item \texttt{description} (\texttt{TEXT}, \texttt{NULL}): Chi tiết lộ trình, yêu cầu cuộn film hay thiết bị mang theo.
        \item \texttt{location} (\texttt{VARCHAR(255)}, \texttt{NOT NULL}): Địa điểm tập trung hoặc địa chỉ tổ chức.
        \item \texttt{event\_date} (\texttt{TIMESTAMPTZ}, \texttt{NOT NULL}): Thời gian bắt đầu diễn ra sự kiện.
        \item \texttt{max\_participants}: Kiểu \texttt{INT}, cho phép \texttt{NULL} (không giới hạn số lượng tham dự) hoặc có ràng buộc \texttt{CHECK (max\_participants > 0)}.
        \item \texttt{created\_at}: Kiểu \texttt{TIMESTAMPTZ}, mặc định \texttt{CURRENT\_TIMESTAMP}.
    \end{itemize}
    \item \textbf{Chiến lược chỉ mục:} Tạo chỉ mục B-tree trên cột \texttt{event\_date} hỗ trợ lọc danh sách sự kiện sắp diễn ra (\texttt{event\_date >= NOW()}) theo thứ tự thời gian tăng dần.
\end{itemize}

\subsubsection{Thiết kế vật lý bảng \texttt{user\_interactions}}

Bảng \texttt{user\_interactions} là bảng ghi nhận nhật ký hành vi người dùng, có tần suất ghi cực kỳ lớn và tăng trưởng kích thước liên tục theo thời gian thực.

\begin{itemize}
    \item \textbf{Khóa chính:} \texttt{interaction\_id} kiểu \texttt{UUID}.
    \item \textbf{Khóa ngoại và quan hệ đa hình (Polymorphic Pattern):}
    \begin{itemize}
        \item \texttt{user\_id} (\texttt{UUID}, \texttt{NOT NULL}): Tham chiếu đến \texttt{users(user\_id)} với tùy chọn \texttt{ON DELETE CASCADE}.
        \item \texttt{target\_type} (\texttt{VARCHAR(50)}, \texttt{NOT NULL}): Xác định phân loại đối tượng tương tác, nhận các giá trị định danh như \texttt{'article'}, \texttt{'film\_lab'}, \texttt{'film\_stock'}.
        \item \texttt{target\_id} (\texttt{UUID}, \texttt{NOT NULL}): Lưu trữ mã định danh của đối tượng được tương tác tương ứng.
    \end{itemize}
    \item \textbf{Thuộc tính hành vi:}
    \begin{itemize}
        \item \texttt{interaction\_type} (\texttt{VARCHAR(20)}, \texttt{NOT NULL}): Loại tương tác của người dùng như \texttt{'view'}, \texttt{'like'}, \texttt{'bookmark'}, \texttt{'share'}.
        \item \texttt{created\_at}: Kiểu \texttt{TIMESTAMPTZ} mặc định \texttt{CURRENT\_TIMESTAMP}.
    \end{itemize}
    \item \textbf{Chiến lược chỉ mục và tối ưu hóa:}
    \begin{itemize}
        \item Tạo chỉ mục phức hợp \texttt{(target\_type, target\_id)} nhằm phục vụ truy vấn tổng hợp lượt tương tác cho một bài viết hoặc lab cụ thể.
        \item Tạo chỉ mục phức hợp \texttt{(user\_id, interaction\_type)} hỗ trợ thuật toán trích xuất danh sách bài viết đã lưu (bookmark) hoặc lịch sử xem của từng cá nhân.
        \item Đối với bảng này ở môi trường sản xuất thực tế, kiến trúc đề xuất áp dụng giải pháp phân vùng theo thời gian (Table Partitioning by Range trên cột \texttt{created\_at}) theo từng tháng hoặc quý để duy trì tốc độ đọc ghi ổn định.
    \end{itemize}
\end{itemize}

\subsubsection{Thiết kế vật lý bảng \texttt{ai\_chat\_log}}

Bảng \texttt{ai\_chat\_log} quản lý lịch sử tương tác giữa người dùng và trợ lý tư vấn thông số nhiếp ảnh AI sử dụng kiến trúc RAG (Retrieval-Augmented Generation).

\begin{itemize}
    \item \textbf{Khóa chính:} \texttt{chat\_id} kiểu \texttt{UUID}.
    \item \textbf{Khóa ngoại:} \texttt{user\_id} dùng kiểu \texttt{UUID} nhưng được phép rỗng (\texttt{NULL}) đi kèm chính sách \texttt{ON DELETE SET NULL}. Điều này cho phép hệ thống lưu vết cả các câu hỏi của khách vãng lai (guest) chưa đăng ký tài khoản, đồng thời vẫn bảo toàn dữ liệu thống kê huấn luyện AI khi một tài khoản người dùng bị xóa.
    \item \textbf{Thuộc tính hội thoại và kiến thức AI:}
    \begin{itemize}
        \item \texttt{user\_query} (\texttt{TEXT}, \texttt{NOT NULL}): Câu hỏi tra cứu kỹ thuật của người dùng.
        \item \texttt{bot\_response} (\texttt{TEXT}, \texttt{NOT NULL}): Phản hồi hướng dẫn hoặc gợi ý thông số film từ mô hình AI.
        \item \texttt{context\_retrieved} (\texttt{TEXT}, \texttt{NULL}): Đoạn trích xuất kiến thức ngữ cảnh (retrieved context chunks) từ kho dữ liệu vector được nạp vào mô hình sinh câu trả lời, phục vụ mục đích kiểm toán (audit) chất lượng câu trả lời.
        \item \texttt{created\_at}: Kiểu \texttt{TIMESTAMPTZ}, mặc định \texttt{CURRENT\_TIMESTAMP}.
    \end{itemize}
    \item \textbf{Chiến lược chỉ mục:} Thiết lập chỉ mục B-tree trên \texttt{user\_id} và \texttt{created\_at} nhằm tải lại giao diện lịch sử hội thoại của người dùng theo thứ tự thời gian một cách nhanh chóng.
\end{itemize}
