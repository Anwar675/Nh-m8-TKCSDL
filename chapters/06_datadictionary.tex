\subsection{Quy tắc toàn vẹn của nhóm P1}

\begin{itemize}
    \item Mỗi email người dùng và mã đơn hàng là duy nhất; mã dịch vụ và mã gói chỉ cần duy nhất trong từng Film Lab.
    \item Dịch vụ, gói dịch vụ và đơn hàng liên quan phải thuộc cùng một Film Lab.
    \item \texttt{total\_amount} bằng tổng tạm tính cộng phí giao nhận và trừ giảm giá; kết quả không được âm.
    \item Giá trong \texttt{order\_items} là giá chốt phục vụ kiểm toán, không thay đổi khi giá danh mục được cập nhật.
    \item Không xóa cứng user, Film Lab hoặc dịch vụ nếu đã phát sinh đơn hàng; sử dụng trạng thái hoặc soft-delete.
\end{itemize}
